* How this works

- This file is picked up by [[https://github.com/nix-community/emacs-overlay][emacs overlay]] and parsed/tangled scanned for ~use-package~-forms
  (https://raw.githubusercontent.com/nix-community/emacs-overlay/master/elisp.nix).
- The required packages are added to the closure
- The tangled blocks are byte-compiled and written to ~/nix/store/...-emacs-default-0.1.0~,
  which is then loaded magically by emacs.

** gptel

See capabilities https://github.com/karthink/gptel.

Precondition: keys in authinfo.

Basic idea:
- Select text, ~SPC p s~ sends it as a prompt
- Select text, ~SPC p r~ sends it as a rewrite request
- Anywhere, ~SPC p p~ opens a new gptel buffer
  - inside it, ~SPC p m~ opens the menu to customize
  - inside a conversation, the org-heading-path to root is the conversation history,
    sibling headings mean conversation-tree branching
  - when inside a conversation, ~SPC p w~ prompts for a conversation name and saves it into ~~/gptel/<conv-name.ai.org~
- opening a ~*.ai.org~ file enables ~gptel-mode~, which stores settings in drawers, making conversations 'reproducible'

#+begin_src elisp
  (use-package gptel
    :preface
    (defvar my/gptel-conversations-directory "~/gptel"
      "Directory for saving gptel conversation files.")

    (defun my/gptel-save-conversation (name)
      "Save current gptel buffer to NAME.ai.org in `my/ai-conversations-directory'."
      (interactive "sConversation name: ")
      (let ((filepath (expand-file-name (concat name ".ai.org")
  				      my/gptel-conversations-directory)))
        (make-directory my/gptel-conversations-directory t)
        (if (derived-mode-p 'org-mode)
  	  (progn (gptel-mode 1)
  		 (write-file filepath))
  	(let ((content (buffer-string)))
  	  (find-file filepath)
  	  (erase-buffer)

  	  (save-buffer)))))

** git-sync

When a repo has a script ~git-sync.sh~ (https://github.com/simonthum/git-sync) in it's root, schedule calling it on save. Debounce the sync to prevent too many commits piling up.

Call it explicitly with bash because the shebang does not seem to work if it's invoked via emacs on termux.

#+begin_src emacs-lisp :lexical t
  (require 'vc-git)

  (defcustom my/git-sync-delay 500
    "Seconds of save-inactivity before running git-sync."
    :type 'integer)

  (defvar my/git-sync--timers (make-hash-table :test #'equal)
    "Pending git-sync timers, keyed by repo root.")

  (defvar my/git-sync--procs (make-hash-table :test #'equal)
    "Running git-sync processes, keyed by repo root.")

  (defun my/git-sync--cancel (root)
    "Cancel any pending git-sync timer for ROOT."
    (when-let* ((old (gethash root my/git-sync--timers)))
      (cancel-timer old))
    (remhash root my/git-sync--timers))

  (defun my/git-sync--schedule (root &optional delay)
    "(Re)start the debounce timer for ROOT."
    (my/git-sync--cancel root)
    (puthash root
  	   (run-with-timer (or delay my/git-sync-delay) nil
  			   #'my/git-sync--run root)
  	   my/git-sync--timers))

  (defun my/git-sync--sentinel (proc _event)
    (when (memq (process-status proc) '(exit signal))
      (let ((root (process-get proc 'root)))
        (remhash root my/git-sync--procs)
        (unless (zerop (process-exit-status proc))
  	(message "git-sync failed (%d) in %s; see %s"
  		 (process-exit-status proc) root
  		 (buffer-name (process-buffer proc)))))))

  (defun my/git-sync--run (root)
    "Run git-sync.sh in ROOT now, or retry shortly if one is already running."
    (my/git-sync--cancel root)
    (let ((script (expand-file-name "git-sync.sh" root)))
      (cond
       ((process-live-p (gethash root my/git-sync--procs))
        (my/git-sync--schedule root 30))
       ((file-exists-p script)
        (let* ((default-directory root)
  	     (process-environment
  	      (cons "GIT_TERMINAL_PROMPT=0" process-environment))
  	     (name (concat "git-sync:" root))
  	     (buf (get-buffer-create (format "*%s*" name))))
  	(with-current-buffer buf
  	  (goto-char (point-max))
  	  (insert (format "\n--- %s ---\n" (current-time-string))))
  	(let ((proc (make-process
  		     :name name
  		     :buffer buf
  		     :command (list "bash" (file-local-name script) "-ns")
  		     :connection-type 'pipe
  		     :file-handler t
  		     :sentinel #'my/git-sync--sentinel)))
  	  (process-put proc 'root root)
  	  (puthash root proc my/git-sync--procs)))))))

  (defun my/git-sync--buffer-root (&optional buffer)
    "Return the repo root of BUFFER's file if it should be git-synced."
    (when-let* ((file (buffer-file-name (or buffer (current-buffer))))
  	      (root (vc-git-root file))
  	      (root (expand-file-name root))
  	      ((file-exists-p (expand-file-name "git-sync.sh" root)))
  	      (git-dir (file-name-as-directory (expand-file-name ".git" root)))
  	      ((not (string-prefix-p git-dir (expand-file-name file)))))
      root))

  (defun my/git-sync--buffer-schedule (&optional buffer)
    "Schedule `bash git-sync.sh -ns' at the repo root of BUFFER's file.
  Runs after `my/git-sync-delay' seconds without further saves."
    (when-let* ((root (my/git-sync--buffer-root buffer)))
      (my/git-sync--schedule root)))

  (defun my/git-sync--buffer-now (&optional buffer)
    "Run `bash git-sync.sh -ns' at the repo root of BUFFER's file immediately."
    (when-let* ((root (my/git-sync--buffer-root buffer)))
      (my/git-sync--run root)))

  (defun my/git-sync-buffer (&optional buffer)
    "Run git-sync in BUFFER's repo, immediately by default.
  With a prefix argument, schedule it instead."
    (interactive)
